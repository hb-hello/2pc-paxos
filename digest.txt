 48:     public void handle(ServerMessage<ClientRequest> request) {
 49:
 50:         if (clientRequestTracker.isAccepted(request)) {
 51:             logger.info("Received duplicate client request {}. Checking for stored reply.", request.getMessageId());
 52:             ServerMessage<ClientReply> replyMessage = clientRequestTracker.getReply(request);
 53:             if (replyMessage != null) {
 54:                 logger.info("Found stored reply for duplicate client request {}. Resending reply.", request.getMessageId());
 55:                 messageSender.sendClientReply(replyMessage);
 56:                 return;
 57:             }
 58:             logger.info("No stored reply found for duplicate client request {}. Ignoring request.", request.getMessageId());
 59:             return;
 60:         }
 61:
 62:         if (!request.payload().getIsReadOnly()) {
 63:             if (paxosServer.isLeader()) handleClientRequestAsLeader(request);
 64:             else {
 65:                 ExecutionMode executionMode = OperationHelper.resolveExecutionMode(serverId, request.payload().getOperation(), accountIdToClusterMap);
 66:                 int otherClusterIndex = OperationHelper.resolveOtherClusterIndex(serverId, request.payload().getOperation(), accountIdToClusterMap);
 67:                 clientRequestTracker.addRequest(request, executionMode, otherClusterIndex);
 68:                 logger.info("Added client request {} to tracker as non-leader with execution mode {} and otherClusterIndex {}", request.getMessageId(), executionMode.name(), otherClusterIndex);
 69:                 paxosServer.handleClientRequestAsNonLeader(request);
 70:             }
 71:
 72:         } else executeReadOnlyAndReply(request);

 79:         if (lockManager.acquireLockAndCheckBalance(clientRequest.getOperation(), executionMode, request.getMessageId(), database)) {
 80:
 81:             if (!clientRequestTracker.hasRequest(request)) {

  1: package org.example.tpc;
  2:
  3: import org.apache.logging.log4j.LogManager;
  4: import org.apache.logging.log4j.Logger;
  5: import org.example.Operation;
  6: import org.example.persistence.KeyValueStore;
  7:
  8: import java.util.*;
  9: import java.util.concurrent.ConcurrentHashMap;
 10: import java.util.concurrent.ConcurrentMap;
 11:
 12: public class LockManager {
 13:     private static final Logger logger = LogManager.getLogger(LockManager.class);
 14:
 15:     // accountId -> owning transactionId
 16:     private final ConcurrentMap<Integer, String> locks = new ConcurrentHashMap<>();
 17:
 18:     /**
 19:      * Try to acquire a lock on accountId for txId.
 20:      * Returns true if:
 21:      *  - the lock was free and is now acquired by txId, or
 22:      *  - the lock was already held by the same txId (idempotent).
 23:      * Returns false if the lock is held by a different transaction.
 24:      */
 25:     public boolean acquireLock(int accountId, String txId) {
 26:         Objects.requireNonNull(txId, "txId must not be null");
 27:
 28:         String existing = locks.putIfAbsent(accountId, txId);
 29:         if (existing == null) {
 30:             return true;
 31:         }
 32:         return existing.equals(txId);
 33:     }
 34:
 35:     /**
 36:      * Release the lock on accountId if it is owned by txId.
 37:      * Returns true if the lock was removed, false otherwise.
 38:      */
 39:     public boolean releaseLock(int accountId, String txId) {
 40:         Objects.requireNonNull(txId, "txId must not be null");
 41:         return locks.remove(accountId, txId);
 42:     }
 43:
 44:     /**
 45:      * Overload: acquire locks based on the Operation and this shard's ExecutionMode.
 46:      * For TRANSFER:
 47:      *   BOTH    -> lock sender and receiver
 48:      *   SENDER  -> lock sender only
 49:      *   RECEIVER-> lock receiver only
 50:      * For BALANCEREQUEST:
 51:      *   lock the requested account.
 52:      *
 53:      * Returns true only if all required locks are acquired; otherwise rolls back and returns false.
 54:      */
 55:     public boolean acquireLock(Operation operation, ExecutionMode mode, String txId) {
 56:         Objects.requireNonNull(operation, "operation must not be null");
 57:         Objects.requireNonNull(mode, "mode must not be null");
 58:         Objects.requireNonNull(txId, "txId must not be null");
 59:
 60:         int[] accountIds = OperationHelper.resolveAccountIds(operation, mode);
 61:         if (accountIds.length == 0) {
 62:             // Nothing to lock (defensive; currently not expected)
 63:             return true;
 64:         }
 65:
 66:         // Deterministic order to reduce deadlock risk if multiple accounts are involved
 67:         Arrays.sort(accountIds);
 68:
 69:         List<Integer> acquired = new ArrayList<>(accountIds.length);
 70:         for (int accountId : accountIds) {
 71:             if (acquireLock(accountId, txId)) {
 72:                 acquired.add(accountId);
 73:             } else {
 74:                 // Failed on one account: release any already-acquired locks for this tx
 75:                 for (int acquiredId : acquired) {
 76:                     releaseLock(acquiredId, txId);
 77:                 }
 78:                 return false;
 79:             }
 80:         }
 81:         return true;
 82:     }
 83:
 84:     /**
 85:      * Acquire all required locks for the given operation and mode, then (for TRANSFER
 86:      * in BOTH or SENDER mode) verify that the sender has sufficient balance.
 87:      *
 88:      * Returns:
 89:      *  - false if locks could not be acquired, or if balance is insufficient;
 90:      *    in both cases any locks acquired for this operation are released.
 91:      *  - true if locks were acquired and (when applicable) balance is sufficient.
 92:      */
 93:     public boolean acquireLockAndCheckBalance(Operation operation,
 94:                                               ExecutionMode mode,
 95:                                               String txId,
 96:                                               KeyValueStore<Double> database) {
 97:         Objects.requireNonNull(operation, "operation must not be null");
 98:         Objects.requireNonNull(mode, "mode must not be null");
 99:         Objects.requireNonNull(txId, "txId must not be null");
100:         Objects.requireNonNull(database, "database must not be null");
101:
102:         // Step 1: try to acquire all relevant locks (sender/receiver depending on mode)
103:         if (!acquireLock(operation, mode, txId)) {
104:             return false;
105:         }
106:
107:         if (mode == ExecutionMode.BOTH) return true; // No balance check needed for intra-shard transfers
108:
109:         // Step 2: only the shard responsible for debiting the sender checks funds
110:         if (operation.getOpCase() == Operation.OpCase.TRANSFER &&
111:                 mode == ExecutionMode.SENDER) {
112:
113:             int senderId = operation.getTransfer().getSender();
114:             double amount = operation.getTransfer().getAmount();
115:
116:             Double currentBalance = database.get(senderId);
117:
118:             boolean sufficient = currentBalance != null && currentBalance >= amount;
119:
120:             if (!sufficient) {
121:                 // This transaction cannot proceed; release locks acquired above.
122:                 releaseLock(operation, mode, txId);
123:                 return false;
124:             }
125:         }
126:
127:         // No funds check needed (e.g. RECEIVER-only transfer or balance request),
128:         // or the check passed.
129:         logger.info("Locks acquired and balance check passed for txId={}", txId);
130:         return true;
131:     }
132:
133:     /**
134:      * Overload: release locks based on the Operation and this shard's ExecutionMode.
135:      * Best-effort: releases any matching (accountId, txId) pairs.
136:      */
137:     public void releaseLock(Operation operation, ExecutionMode mode, String txId) {
138:         Objects.requireNonNull(operation, "operation must not be null");
139:         Objects.requireNonNull(mode, "mode must not be null");
140:         Objects.requireNonNull(txId, "txId must not be null");
141:
142:         int[] accountIds = OperationHelper.resolveAccountIds(operation, mode);
143:         for (int accountId : accountIds) {
144:             releaseLock(accountId, txId);
145:         }
146:     }
147:
148:     /**
149:      * Best-effort release of all locks held by txId.
150:      */
151:     public void releaseAllForTransaction(String txId) {
152:         Objects.requireNonNull(txId, "txId must not be null");
153:         locks.entrySet().removeIf(e -> txId.equals(e.getValue()));
154:     }
155:
156:     /**
157:      * Release all currently held locks on this node.
158:      * Intended to be called when the node transitions to a backup role
159:      * and should not own any application-level locks.
160:      */
161:     public void releaseAllLocks() {
162:         locks.clear();
163:     }
164: }

 18:     private static final Logger logger = LogManager.getLogger(MessageRetryManager.class);
 19:

 21:         COMMIT,
 22:         ABORT
 23:     }
 24:
 25:     private final TPCMessageSender messageSender;
 26:     private final long intervalMillis;
 27:     private final ScheduledExecutorService scheduler;
 28:
 29:     // One retry loop per messageId
 30:     private final ConcurrentMap<String, ScheduledFuture<?>> tasks = new ConcurrentHashMap<>();
 31:     private final ConcurrentMap<String, Boolean> firstSendCommitDone = new ConcurrentHashMap<>();
 32:     private final ConcurrentMap<String, Boolean> firstSendAbortDone = new ConcurrentHashMap<>();
 33:
 34:     public MessageRetryManager(TPCMessageSender messageSender,
 35:                                long intervalMillis,
 36:                                ScheduledExecutorService scheduler) {
 37:         this.messageSender = Objects.requireNonNull(messageSender, "messageSender");
 38:         this.intervalMillis = intervalMillis;
 39:         this.scheduler = Objects.requireNonNull(scheduler, "scheduler");
 40:     }
 41:
 42:     // ---------- COMMIT flow ----------
 43:
 44:     public void startCommitRetries(int targetNodeId,
 45:                                    ServerMessage<TPCCommitMessage> commit) {
 46:         String messageId = commit.getMessageId();
 47:
 48:         // Ensure we only create one task per messageId, even if multiple threads call this.
 49:         firstSendCommitDone.putIfAbsent(messageId, Boolean.FALSE);
 50:
 51:         Runnable sendTask = () -> {
 52:             // If we were cancelled concurrently, just exit.
 53:             if (!tasks.containsKey(messageId)) {
 54:                 return;
 55:             }
 56:
 57:             boolean alreadySentFirst =
 58:                     firstSendCommitDone.getOrDefault(messageId, Boolean.FALSE);
 59:
 60:             logger.info("Retrying COMMIT for messageId={} (firstSendDone={})",
 61:                     messageId, alreadySentFirst);
 62:
 63:             StreamObserver<TPCAckMessage> observer =
 64:                     createAckObserver(MessageType.COMMIT, messageId);
 65:
 66:             if (!alreadySentFirst) {
 67:                 messageSender.sendCommit(targetNodeId, commit, observer);
 68:                 firstSendCommitDone.put(messageId, Boolean.TRUE);
 69:             } else {
 70:                 messageSender.broadcastCommitToCluster(targetNodeId, commit, observer);
 71:             }
 72:         };
 73:
 74:         ScheduledFuture<?> newFuture =
 75:                 scheduler.scheduleAtFixedRate(sendTask, 0, intervalMillis, TimeUnit.MILLISECONDS);
 76:
 77:         // Atomically register the task if absent; cancel the new one if some other thread won.
 78:         ScheduledFuture<?> existing = tasks.putIfAbsent(messageId, newFuture);
 79:         if (existing != null) {
 80:             newFuture.cancel(false);
 81:         }
 82:     }
 83:
 84:     public void stopCommitRetries(ServerMessage<TPCCommitMessage> commit) {
 85:         stopRetries(MessageType.COMMIT, commit.getMessageId());
 86:     }
 87:
 88:     // ---------- ABORT flow ----------
 89:
 90:     public void startAbortRetries(int targetNodeId,
 91:                                   ServerMessage<TPCAbortMessage> abort) {
 92:         String messageId = abort.getMessageId();
 93:
 94:         firstSendAbortDone.putIfAbsent(messageId, Boolean.FALSE);
 95:
 96:         Runnable sendTask = () -> {
 97:             if (!tasks.containsKey(messageId)) {
 98:                 return;
 99:             }
100:
101:             boolean alreadySentFirst =
102:                     firstSendAbortDone.getOrDefault(messageId, Boolean.FALSE);
103:
104:             logger.info("Retrying ABORT for messageId={} (firstSendDone={})",
105:                     messageId, alreadySentFirst);
106:
107:             StreamObserver<TPCAckMessage> observer =
108:                     createAckObserver(MessageType.ABORT, messageId);
109:
110:             if (!alreadySentFirst) {
111:                 messageSender.sendAbort(targetNodeId, abort, observer);
112:                 firstSendAbortDone.put(messageId, Boolean.TRUE);
113:             } else {
114:                 messageSender.broadcastAbortToCluster(targetNodeId, abort, observer);
115:             }
116:         };
117:
118:         ScheduledFuture<?> newFuture =
119:                 scheduler.scheduleAtFixedRate(sendTask, 0, intervalMillis, TimeUnit.MILLISECONDS);
120:
121:         ScheduledFuture<?> existing = tasks.putIfAbsent(messageId, newFuture);
122:         if (existing != null) {
123:             newFuture.cancel(false);
124:         }
125:     }
126:
127:     public void stopAbortRetries(ServerMessage<TPCAbortMessage> abort) {
128:         stopRetries(MessageType.ABORT, abort.getMessageId());
129:     }
130:
131:     // ---------- Common helpers ----------
132:
133:     private StreamObserver<TPCAckMessage> createAckObserver(MessageType type,
134:                                                             String messageId) {
135:         return new StreamObserver<>() {
136:             @Override
137:             public void onNext(TPCAckMessage value) {
138:                 logger.info("Received TPCAck for {} messageId={}", type, messageId);
139:                 stopRetries(type, messageId);
140:             }
141:
142:             @Override
143:             public void onError(Throwable t) {
144:                 logger.warn("{} retry failed for messageId={}: {}",
145:                         type, messageId, t.getMessage());
146:                 // keep retrying; do not cancel here
147:             }
148:
149:             @Override
150:             public void onCompleted() {
151:                 // no-op
152:             }
153:         };
154:     }
155:
156:     private void stopRetries(MessageType type, String messageId) {
157:         // Remove and cancel atomically; safe if called concurrently with start/retry.
158:         ScheduledFuture<?> fut = tasks.remove(messageId);
159:         if (fut != null) {
160:             fut.cancel(false);
161:         }
162:         if (type == MessageType.COMMIT) {
163:             firstSendCommitDone.remove(messageId);
164:         } else {
165:             firstSendAbortDone.remove(messageId);
166:         }
167:     }
168:
169:     public void shutdown() {
170:         for (Map.Entry<String, ScheduledFuture<?>> e : tasks.entrySet()) {
171:             ScheduledFuture<?> fut = tasks.remove(e.getKey());
172:             if (fut != null) {
173:                 fut.cancel(false);
174:             }
175:         }
176:         firstSendCommitDone.clear();
177:         firstSendAbortDone.clear();
178:     }
179: }

19:                                                      Operation operation,
20:                                                      Map<Integer, Integer> database) {
21:         try {
22:             int localCluster = Config.getServerClusterIndex(serverId);
23:             if (operation.hasTransfer()) {

33:                 } else if (senderLocal) {
34:                     return ExecutionMode.SENDER;
35:                 } else if (receiverLocal) {
36:                     return ExecutionMode.RECEIVER;

46:             return ExecutionMode.BOTH;

54:             return -1;
55:         }
56:         int localCluster = Config.getServerClusterIndex(serverId);
57:         Transfer transfer = operation.getTransfer();
58:         Integer senderCluster = database.get(transfer.getSender());
59: //        logger.info("Sender {} is in cluster {}", transfer.getSender(), senderCluster);
60:         Integer receiverCluster = database.get(transfer.getReceiver());
61: //        logger.info("Receiver {} is in cluster {}", transfer.getReceiver(), receiverCluster);

 84:     /**
 85:      * Asynchronously execute committed client requests from the Paxos operation log,
 86:      * in-order and at-most-once, up to and including seqNum.
 87:      * <p>
 88:      * This method is safe to call repeatedly / concurrently; all work is serialized
 89:      * on the stateExecutor.
 90:      */
 91:     public CompletableFuture<Void> execute(long seqNum, ExecutionMode mode) {
 92:         CompletableFuture<Void> future = new CompletableFuture<>();
 93:         stateMachineExecutor.execute(() -> {
 94:             try {
 95:                 executeUpTo(seqNum, mode);
 96:                 future.complete(null);
 97:             } catch (Throwable t) {
 98:                 logger.error("Error executing up to seqNum {}", seqNum, t);
 99:                 future.completeExceptionally(t);
100:             }
101:         });
102:         return future;
103:     }
104:
105:     private void executeUpTo(long targetSeq, ExecutionMode mode) {
106:         while (true) {
107:             long current = nextToExecute.get();
108:             if (current > targetSeq) {
109:                 return;
110:             }
111:
112:             ServerMessage<ClientRequest> requestMessage = operationLog.getRequest(current);
113:             if (requestMessage == null) {
114:                 return;
115:             }
116:             ClientRequest request = requestMessage.payload();
117:             String requestId = requestMessage.getMessageId();
118:
119:             // Skip if already marked committed/executed
120:             if (Boolean.TRUE.equals(committedOrAborted.get(requestId))) {
121:                 nextToExecute.incrementAndGet();
122:                 continue;
123:             }
124:
125:             OperationStatus status = operationLog.getStatus(current); // NONE/PREPARED/COMMITTED/EXECUTED/CHECKPOINTED
126:             Phase phase = operationLog.getEntry(current).phase();
127:             if (status != OperationStatus.COMMITTED && status != OperationStatus.EXECUTED && (phase == Phase.PREPARE || phase == Phase.INTRA_SHARD)) {
128:                 // Not ready to execute in log order yet.
129:                 return;
130:             }
131:
132:             if (status == OperationStatus.EXECUTED) {
133:                 nextToExecute.incrementAndGet();
134:                 continue;
135:             }
136:
137:             // Build WAL before applying the operation
138:             WalEntry entry = buildWalEntry(request, mode);
139:             if (entry != null) {
140:                 wal.put(requestId, entry);
141:             }
142:
143:             // Execute the operation on the state machine
144:             OperationResult result =
145:                     stateMachine.execute(request.getOperation(), mode);
146:
147:             logger.info("Executed seq {} for client {} resultCase={}",
148:                     current, request.getClientId(), result.getResultCase());
149:
150:             // Build and store client reply in the tracker
151:             ClientReply reply = ClientReply.newBuilder()
152:                     .setResult(result)
153:                     .setSenderId(serverId)
154:                     .setClientId(request.getClientId())
155:                     .setTimestamp(request.getTimestamp())
156:                     .build();
157:             ServerMessage<ClientReply> replyMessage = new ServerMessage<>(reply);
158:             requestTracker.storeReply(requestMessage, replyMessage);
159:
160:             // Mark as EXECUTED in the Paxos log atomically to ensure at-most-once
161:             operationLog.compareAndSetStatus(
162:                     current, OperationStatus.COMMITTED, OperationStatus.EXECUTED);
163:
164:             nextToExecute.incrementAndGet();
165:
166:             // Notify listener that the operation has been executed
167:             onExecuted.accept(requestMessage, operationLog.getEntry(current).phase());
168:         }
169:     }

171:     /**
172:      * For non-read-only operations, capture the pre-operation balances of all
173:      * accounts that may be mutated, as a before-image.
174:      */
175:     private WalEntry buildWalEntry(ClientRequest request, ExecutionMode mode) {
176:         Operation op = request.getOperation();
177:         switch (op.getOpCase()) {
178:             case TRANSFER -> {
179:                 int sender = op.getTransfer().getSender();
180:                 int receiver = op.getTransfer().getReceiver();
181:                 Map<Integer, Double> before = new HashMap<>(2);
182:                 if (mode == ExecutionMode.BOTH || mode == ExecutionMode.SENDER) {
183:                     before.put(sender, stateMachineBalance(sender));
184:                 }
185:                 if (mode == ExecutionMode.BOTH || mode == ExecutionMode.RECEIVER) {
186:                     before.put(receiver, stateMachineBalance(receiver));
187:                 }
188:                 if (before.isEmpty()) {
189:                     return null;
190:                 }
191:                 return new WalEntry(before);
192:             }
193:             case BALANCE_REQUEST -> {
194:                 return null;
195:             }
196:             case OP_NOT_SET -> throw new IllegalArgumentException("Operation.op not set");
197:         }
198:         throw new IllegalStateException("Unhandled opCase: " + op.getOpCase());
199:     }

206:                 ).build();

208:                 stateMachine.execute(balanceOp, ExecutionMode.BOTH);
209:         return res.getBalance();

218:         stateMachineExecutor.execute(() -> {
219:             try {

226:                     }
227:                 }

229:                 f.complete(null);
230:             } catch (Throwable t) {
231:                 logger.error("Error undoing request: {}", requestId, t);
232:                 f.completeExceptionally(t);

 12:     private static final Logger logger = LogManager.getLogger(LivenessTimer.class);
 13:
 14:     private final long timeoutMillis;
 15:     private final ScheduledExecutorService scheduler;
 16:     private final ExecutorService callbackExecutor;  // Separate executor for callbacks
 17:     private final Runnable callback;
 18:     private final AtomicReference<ScheduledFuture<?>> scheduledFutureRef;
 19:     private final AtomicBoolean isRunning;
 20:     private final AtomicBoolean isExecutingCallback;
 21:     private final AtomicLong runGeneration;
 22:
 23:     public LivenessTimer(long timeoutMillis, Runnable callback) {
 24:         this.timeoutMillis = timeoutMillis;
 25:         this.callback = callback;
 26:         this.scheduler = Executors.newSingleThreadScheduledExecutor();
 27:         this.callbackExecutor = Executors.newSingleThreadExecutor();  // Dedicated thread for callbacks
 28:         this.scheduledFutureRef = new AtomicReference<>();
 29:         this.isRunning = new AtomicBoolean(false);
 30:         this.isExecutingCallback = new AtomicBoolean(false);
 31:         this.runGeneration = new AtomicLong(0L);
 32:     }
 33:
 34:     private void start(String name) {
 35:         try {
 36:             isRunning.set(true);
 37:             final long runId = runGeneration.incrementAndGet();
 38:             ScheduledFuture<?> future = scheduler.schedule(() -> {
 39:                 // Timer has fired - no longer running
 40:                 isRunning.set(false);
 41:                 if (runGeneration.get() != runId) {
 42:                     logger.debug("Timer firing skipped for {} due to stale run {}", name, runId);
 43:                     return;
 44:                 }
 45:                 // Submit callback to separate executor
 46:                 callbackExecutor.submit(() -> {
 47:                     if (runGeneration.get() != runId) {
 48:                         return;
 49:                     }
 50:                     isExecutingCallback.set(true);
 51:                     try {
 52:                         logger.info("Timer callback executing (isRunning=false) for : {}", name);
 53:                         callback.run();
 54:                     } catch (Exception e) {
 55:                         logger.error("Error in timer callback: {}", e.getMessage());
 56:                     } finally {
 57:                         isExecutingCallback.set(false);
 58:                     }
 59:                 });
 60:             }, timeoutMillis, TimeUnit.MILLISECONDS);
 61:
 62:             scheduledFutureRef.set(future);
 63:         } catch (Exception e) {
 64:             throw new RuntimeException("Timer errored out : " + e.getMessage());
 65:         }
 66:     }

 68:     public void startIfNotRunning(String name) {
 69:         if (!isRunning.get()) {
 70:             start(name);
 71:         }
 72:     }
 73:
 74:     public void restart(String name) {
 75:         stop();
 76:         start(name);
 77:     }
 78:
 79:     public synchronized void stop() {
 80:         ScheduledFuture<?> currentFuture = scheduledFutureRef.get();
 81:         if (currentFuture != null) {
 82:             boolean mayInterrupt = !isExecutingCallback.get();
 83:             currentFuture.cancel(mayInterrupt);
 84:             scheduledFutureRef.set(null);
 85:         }
 86:         isRunning.set(false);
 87:         runGeneration.incrementAndGet();
 88:     }
 89:
 90:     public boolean isRunning() {
 91:         return isRunning.get();
 92:     }
 93:
 94:     public boolean isExecutingCallback() {
 95:         return isExecutingCallback.get();
 96:     }
 97:
 98:     public void shutdown() {
 99:         stop();
100:
101:         // Shutdown scheduler
102:         scheduler.shutdown();
103:         try {
104:             if (!scheduler.awaitTermination(5, TimeUnit.SECONDS)) {
105:                 scheduler.shutdownNow();
106:             }
107:         } catch (InterruptedException e) {
108:             scheduler.shutdownNow();

// filename: ServerMessageTracker.java
  1: package org.example.state;
  2:
  3: import com.google.protobuf.MessageLite;
  4: import org.example.messaging.ServerMessage;
  5:
  6: import java.util.Objects;
  7: import java.util.concurrent.ConcurrentHashMap;
  8: import java.util.concurrent.ConcurrentMap;
  9: import java.util.concurrent.atomic.AtomicBoolean;
 10: import java.util.concurrent.atomic.AtomicInteger;
 11:
 12: /**
 13:  * Tracks all ServerMessage instances seen by this Paxos node.
 14:  * - De-duplicates by messageId.
 15:  * - Tracks consensus by messageIndex with quorum checks.
 16:  *
 17:  * Thread-safe for concurrent access.
 18:  */
 19: public final class ServerMessageTracker {
 20:
 21:     // messageId -> ServerMessage (for de-duplication and optional inspection)
 22:     private final ConcurrentMap<String, ServerMessage<? extends MessageLite>> messages =
 23:             new ConcurrentHashMap<>();
 24:
 25:     // messageIndex -> vote counter + quorum flag
 26:     private final ConcurrentMap<String, ConsensusCounter> consensusByIndex =
 27:             new ConcurrentHashMap<>();
 28:
 29:     private static final class ConsensusCounter {
 30:         private final AtomicInteger votes = new AtomicInteger(0);
 31:         private final AtomicBoolean quorumSignalled = new AtomicBoolean(false);
 32:
 33:         /**
 34:          * Increment votes and return true ONLY the first time
 35:          * votes >= quorumNeeded and this method is called.
 36:          */
 37:         boolean addVoteAndMaybeSignal(int quorumNeeded) {
 38:             int v = votes.incrementAndGet();
 39:             if (v >= quorumNeeded) {
 40:                 // Only one thread flips from false -> true.
 41:                 return quorumSignalled.compareAndSet(false, true);
 42:             }
 43:             return false;
 44:         }
 45:
 46:         /**
 47:          * Read-only check: has quorum been reached (now or earlier)?
 48:          */
 49:         boolean hasQuorum(int quorumNeeded) {
 50:             return quorumSignalled.get() || votes.get() >= quorumNeeded;
 51:         }
 52:     }
 53:
 54:     public boolean addMessage(ServerMessage<? extends MessageLite> message) {
 55:         Objects.requireNonNull(message, "message");
 56:         String id = message.getMessageId(); // from updated ServerMessage [attached_file:1]
 57:         ServerMessage<? extends MessageLite> existing = messages.putIfAbsent(id, message);
 58:         return existing == null;
 59:     }
 60:
 61:     public boolean hasMessage(ServerMessage<? extends MessageLite> message) {
 62:         Objects.requireNonNull(message, "message");
 63:         return hasMessage(message.getMessageId());
 64:     }
 65:
 66:     public boolean hasMessage(String messageId) {
 67:         Objects.requireNonNull(messageId, "messageId");
 68:         return messages.containsKey(messageId);
 69:     }
 70:
 71:     /**
 72:      * Add a message and update consensus state for its messageIndex.
 73:      *
 74:      * Semantics:
 75:      * - Returns true exactly ONCE per messageIndex, the first time quorum is reached.
 76:      * - Returns false for:
 77:      *   - duplicates (same messageId),
 78:      *   - additional distinct messages after quorum is already reached.
 79:      */
 80:     public boolean addMessageWithConsensus(ServerMessage<? extends MessageLite> message,
 81:                                            int quorumNeeded) {
 82:         Objects.requireNonNull(message, "message");
 83:
 84:         // 1) Never signal on duplicates.
 85:         if (hasMessage(message)) {
 86:             return false;
 87:         }
 88:
 89:         // 2) Try to record this messageId. If a race just inserted it, treat as duplicate.
 90:         boolean added = addMessage(message);
 91:         if (!added) {
 92:             return false;
 93:         }
 94:
 95:         // 3) New unique messageId; count it toward consensus by messageIndex.
 96:         String index = message.getMessageIndex();
 97:         ConsensusCounter counter = consensusByIndex
 98:                 .computeIfAbsent(index, k -> new ConsensusCounter());
 99:
100:         // Returns true only for the first thread that both reaches quorum
101:         // and flips quorumSignalled from false -> true.
102:         return counter.addVoteAndMaybeSignal(quorumNeeded);
103:     }
104:
105:     /**
106:      * Read-only check: has this index reached quorum (possibly earlier)?
107:      */
108:     public boolean quorumCheck(String messageIndex, int quorumNeeded) {
109:         Objects.requireNonNull(messageIndex, "messageIndex");
110:         ConsensusCounter counter = consensusByIndex.get(messageIndex);
111:         if (counter == null) {
112:             return false;
113:         }
114:         return counter.hasQuorum(quorumNeeded);
115:     }
116: }
